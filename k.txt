apiVersion: apps/v1
kind: Deployment

metadata:
  name: tomcat-web-deployment

spec:
  replicas: 3

  selector:
    matchLabels:
      app: tomcat-web

  template:
    metadata:
      labels:
        app: tomcat-web

    spec:
      containers:
        - name: tomcat-web
          image: tomcat-web-app:latest
          imagePullPolicy: IfNotPresent

          ports:
            - containerPort: 8080

apiVersion: v1
kind: Service

metadata:
  name: tomcat-web-service

spec:
  type: NodePort

  selector:
    app: tomcat-web

  ports:
    - protocol: TCP
      port: 8080
      targetPort: 8080
      nodePort: 30080
